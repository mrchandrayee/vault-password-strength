function analyzePassword(value) {
  const pool = getCharacterPool(value);
  const entropy =
    value.length * Math.log2(pool);
  return classify(entropy);
}

password.addEventListener(
  "input", event => {
    render(analyzePassword(
      event.target.value
    ));
  }
);
